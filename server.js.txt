function encrypt(text) {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv('aes-256-gcm', Buffer.from(ENCRYPTION_KEY_HEX, 'hex'), iv);
  let encrypted = cipher.update(text, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  const tag = cipher.getAuthTag().toString('hex');
  return iv.toString('hex') + ':' + tag + ':' + encrypted;
}
function decrypt(encStr) {
  const parts = encStr.split(':');
  const ivHex = parts[0]; const tagHex = parts[1]; const data = parts[2];
  const decipher = crypto.createDecipheriv('aes-256-gcm', Buffer.from(ENCRYPTION_KEY_HEX, 'hex'), Buffer.from(ivHex, 'hex'));
  decipher.setAuthTag(Buffer.from(tagHex, 'hex'));
  let dec = decipher.update(data, 'hex', 'utf8');
  dec += decipher.final('utf8');
  return dec;
}

const TWITCH_OAUTH_AUTHORIZE = 'https://id.twitch.tv/oauth2/authorize';
const TWITCH_OAUTH_TOKEN = 'https://id.twitch.tv/oauth2/token';
const TWITCH_HELIX = 'https://api.twitch.tv/helix';
const EVENTSUB_WS_URL = 'wss://eventsub.wss.twitch.tv/ws';

const REQUIRED_SCOPES = [
  'moderator:read:followers',
  'channel:read:subscriptions',
  'bits:read',
  'user:read:email',
];

async function refreshAccessToken(refreshToken) {
  const res = await axios.post(TWITCH_OAUTH_TOKEN, null, {
    params: {
      client_id: TWITCH_CLIENT_ID,
      client_secret: TWITCH_CLIENT_SECRET,
      grant_type: 'refresh_token',
      refresh_token: refreshToken,
    }
  });
  return res.data;
}
async function getTwitchUser(accessToken) {
  const res = await axios.get(TWITCH_HELIX + '/users', {
    headers: { 'Client-Id': TWITCH_CLIENT_ID, 'Authorization': 'Bearer ' + accessToken }
  });
  return res.data.data[0];
}
function getUserTokens(userId) {
  const u = users.get(userId);
  if (!u) return null;
  return { accessToken: decrypt(u.accessTokenEnc), refreshToken: decrypt(u.refreshTokenEnc) };
}

function triggerEvent(userId, event) {
  const user = users.get(userId);
  if (!user) return;
  const overlayIds = user.overlayIds || new Set();
  for (const overlayId of overlayIds) {
    const clients = appWsClients.get(overlayId);
    if (!clients) continue;
    const payload = JSON.stringify({
      ...event,
      overlayId: overlayId,
      timestamp: new Date().toISOString(),
      config: (overlays.get(overlayId) && overlays.get(overlayId).config) || null,
    });
    for (const ws of clients) {
      if (ws.readyState === WebSocket.OPEN) ws.send(payload);
    }
  }
  console.log('[triggerEvent] user=' + (user.login || userId) + ' overlays=' + overlayIds.size + ' type=' + event.type + ' username=' + event.username);
}

async function connectEventSubForUser(userId) {
  const existing = eventSubConnections.get(userId);
  if (existing && existing.ws && existing.ws.readyState === WebSocket.OPEN) return existing;

  const tokens = getUserTokens(userId);
  if (!tokens) throw new Error('Tokens introuvables');
  let twitchUser;
  try { twitchUser = await getTwitchUser(tokens.accessToken); }
  catch (e) {
    const refreshed = await refreshAccessToken(tokens.refreshToken);
    const u = users.get(userId);
    u.accessTokenEnc = encrypt(refreshed.access_token);
    u.refreshTokenEnc = encrypt(refreshed.refresh_token);
    users.set(userId, u);
    twitchUser = await getTwitchUser(refreshed.access_token);
  }
  const broadcasterId = twitchUser.id;
  console.log('[EventSub] Connexion pour ' + twitchUser.login + ' (' + broadcasterId + ')');
  const ws = new WebSocket(EVENTSUB_WS_URL);
  const conn = { ws: ws, sessionId: null, broadcasterId: broadcasterId, userId: userId, subscriptions: new Map(), reconnectUrl: null };
  eventSubConnections.set(userId, conn);

  ws.on('open', function(){ console.log('[EventSub WS open] ' + userId); });
  ws.on('message', async function(data){
    try {
      const msg = JSON.parse(data.toString());
      const type = msg.metadata && msg.metadata.message_type;
      if (type === 'session_welcome') {
        const sessionId = msg.payload.session.id;
        conn.sessionId = sessionId;
        console.log('[EventSub welcome] sessionId=' + sessionId + ' pour ' + userId);
        await createEventSubSubscriptions(userId, sessionId, broadcasterId);
      } else if (type === 'session_keepalive') {
      } else if (type === 'notification') {
        const subType = msg.payload.subscription.type;
        const event = msg.payload.event;
        console.log('[EventSub notif] ' + subType + ' -> ' + (event.user_name || event.from_broadcaster_user_name || '?'));
        handleEventSubNotification(userId, subType, event);
      } else if (type === 'session_reconnect') {
        const newUrl = msg.payload.session.reconnect_url;
        console.log('[EventSub reconnect] vers ' + newUrl);
        conn.reconnectUrl = newUrl;
        ws.close();
        setTimeout(function(){ connectEventSubForUser(userId).catch(console.error); }, 1000);
      } else if (type === 'revocation') {
        console.warn('[EventSub revocation] ' + msg.payload.subscription.type + ' pour ' + userId);
      }
    } catch(e){ console.error('[EventSub message error]', e); }
  });
  ws.on('close', function(code){
    console.log('[EventSub WS close] ' + userId + ' code=' + code);
    if (!conn.reconnectUrl) {
      setTimeout(function(){ if(users.has(userId)) connectEventSubForUser(userId).catch(console.error); }, 5000);
    }
  });
  ws.on('error', function(e){ console.error('[EventSub WS error] ' + userId, e.message); });
  return conn;
}

async function createEventSubSubscriptions(userId, sessionId, broadcasterUserId) {
  const tokens = getUserTokens(userId);
  const headers = { 'Client-Id': TWITCH_CLIENT_ID, 'Authorization': 'Bearer ' + tokens.accessToken, 'Content-Type': 'application/json' };
  const subsToCreate = [
    { type: 'channel.follow', version: '2', condition: { broadcaster_user_id: broadcasterUserId, moderator_user_id: broadcasterUserId } },
    { type: 'channel.subscribe', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
    { type: 'channel.subscription.gift', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
    { type: 'channel.subscription.message', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
    { type: 'channel.cheer', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
    { type: 'channel.raid', version: '1', condition: { to_broadcaster_user_id: broadcasterUserId } },
    { type: 'stream.online', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
    { type: 'stream.offline', version: '1', condition: { broadcaster_user_id: broadcasterUserId } },
  ];
  for (const sub of subsToCreate) {
    try {
      const res = await axios.post(TWITCH_HELIX + '/eventsub/subscriptions', {
        type: sub.type, version: sub.version, condition: sub.condition,
        transport: { method: 'websocket', session_id: sessionId }
      }, { headers: headers });
      const conn = eventSubConnections.get(userId);
      if (conn) conn.subscriptions.set(sub.type, res.data.data[0].id);
      console.log('[EventSub sub OK] ' + sub.type + ' pour ' + userId);
    } catch (e) {
      const errData = e.response && e.response.data;
      console.error('[EventSub sub FAIL] ' + sub.type + ' pour ' + userId + ': ' + (errData && errData.message || e.message));
    }
  }
}

function handleEventSubNotification(userId, subType, event) {
  let normalized = null;
  switch(subType){
    case 'channel.follow': normalized = { type: 'follow', username: event.user_name, userId: event.user_id }; break;
    case 'channel.subscribe': normalized = { type: 'sub', username: event.user_name, tier: event.tier, isGift: event.is_gift }; break;
    case 'channel.subscription.gift': normalized = { type: 'gift_sub', username: event.user_name, amount: event.total, tier: event.tier, isAnonymous: event.is_anonymous, cumulative: event.cumulative_total }; break;
    case 'channel.subscription.message': normalized = { type: 'resub', username: event.user_name, tier: event.tier, message: event.message && event.message.text, cumulativeMonths: event.cumulative_months, streakMonths: event.streak_months }; break;
    case 'channel.cheer': normalized = { type: 'bits', username: event.user_name, amount: event.bits, message: event.message }; break;
    case 'channel.raid': normalized = { type: 'raid', username: event.from_broadcaster_user_name, viewers: event.viewers, fromId: event.from_broadcaster_user_id }; break;
    case 'stream.online': normalized = { type: 'stream_online', username: event.broadcaster_user_name }; break;
    case 'stream.offline': normalized = { type: 'stream_offline', username: event.broadcaster_user_name }; break;
    default: return;
  }
  if (normalized) { normalized.raw = event; triggerEvent(userId, normalized); }
}

app.get('/auth/twitch', function(req,res){
  const state = crypto.randomBytes(16).toString('hex');
  req.session.oauthState = state;
  const params = new URLSearchParams({ client_id: TWITCH_CLIENT_ID, redirect_uri: TWITCH_REDIRECT_URI, response_type: 'code', scope: REQUIRED_SCOPES.join(' '), state: state, force_verify: 'false' });
  res.redirect(TWITCH_OAUTH_AUTHORIZE + '?' + params.toString());
});
app.get('/auth/twitch/callback', async function(req,res){
  const code = req.query.code; const state = req.query.state;
  if (!code || state !== req.session.oauthState) return res.status(400).send('Invalid state or code');
  try {
    const tokenRes = await axios.post(TWITCH_OAUTH_TOKEN, null, { params: { client_id: TWITCH_CLIENT_ID, client_secret: TWITCH_CLIENT_SECRET, code: code, grant_type: 'authorization_code', redirect_uri: TWITCH_REDIRECT_URI } });
    const access_token = tokenRes.data.access_token; const refresh_token = tokenRes.data.refresh_token;
    const twitchUser = await getTwitchUser(access_token);
    const userId = twitchUser.id;
    const overlayId = 'overlay_' + userId + '_' + crypto.randomBytes(4).toString('hex');
    const existing = users.get(userId);
    const overlaySet = existing && existing.overlayIds ? existing.overlayIds : new Set();
    overlaySet.add(overlayId);
    users.set(userId, { twitchUserId: twitchUser.id, login: twitchUser.login, displayName: twitchUser.display_name, profileImage: twitchUser.profile_image_url, accessTokenEnc: encrypt(access_token), refreshTokenEnc: encrypt(refresh_token), overlayIds: overlaySet, createdAt: (existing && existing.createdAt) || new Date().toISOString() });
    overlays.set(overlayId, { userId: userId, config: { theme: 'neon', position: 'center', animation: 'bounce', duration: 5000, sound: true }, createdAt: new Date().toISOString() });
    req.session.userId = userId; req.session.twitchLogin = twitchUser.login;
    connectEventSubForUser(userId).catch(function(e){ console.error('EventSub connect fail', e.message); });
    const frontendRedirect = FRONTEND_URL + '/dashboard?overlayId=' + overlayId + '&connected=1';
    res.redirect(frontendRedirect);
  } catch(e){ console.error('OAuth callback error', e.response && e.response.data || e.message); res.status(500).send('OAuth error: ' + (e.response && e.response.data && e.response.data.message || e.message)); }
});

const server = http.createServer(app);
const wss = new WebSocketServer({ noServer: true });
server.on('upgrade', function(req,socket,head){
  const url = new URL(req.url, 'http://' + req.headers.host);
  if (url.pathname.startsWith('/ws/overlay/')) {
    const overlayId = url.pathname.split('/').pop();
    if (!overlays.has(overlayId)) { socket.destroy(); return; }
    wss.handleUpgrade(req,socket,head,function(ws){ wss.emit('connection', ws, req, overlayId); });
  } else { socket.destroy(); }
});
wss.on('connection', function(ws, req, overlayId){
  if (!appWsClients.has(overlayId)) appWsClients.set(overlayId, new Set());
  appWsClients.get(overlayId).add(ws);
  console.log('[App WS] overlay ' + overlayId + ' client connecte (' + appWsClients.get(overlayId).size + ')');
  ws.on('close', function(){
    const set = appWsClients.get(overlayId);
    if (set){ set.delete(ws); if (set.size===0) appWsClients.delete(overlayId); }
    console.log('[App WS] overlay ' + overlayId + ' deconnecte');
  });
});
